cmake_minimum_required(VERSION 3.20)
project(DepthReplay VERSION 0.1.0 LANGUAGES CXX)
add_executable(depthreplay src/main.cpp)
target_compile_features(depthreplay PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(depthreplay PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(depthreplay PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
